    wait NULL

    Memory.Allocate(STRING_SIZE, STRING_BUFFER)
    Text.StringFormat(STRING_BUFFER, "Cutscene time: %d", curtime) // display_text_WITH_NUMBER {leftTopX} 10.0 {leftTopY} 50.0 {key} 'CTVCT' {value} curtime

    Text.StringFormat(STRING_BUFFER, "Cutscene ID: %d", cutscene_id) // display_text_WITH_NUMBER {leftTopX} 10.0 {leftTopY} 60.0 {key} 'CTVID' {value} cutscene_id
    Text.DrawString(STRING_BUFFER, DrawEvent.AfterFade, 10.0, 60.0, SIZE_X, SIZE_Y, true, Font.Subtitles)

    Memory.Allocate(STRING_SIZE, STRING_BUFFER)

    Memory.Free(STRING_BUFFER)

end
